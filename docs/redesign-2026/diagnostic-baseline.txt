<script is:inline>
  window.bescheibenTurnstileSuccess = (token) => {
    const input = document.querySelector('[data-turnstile-token]');
    if (input) input.value = token;
  };
  window.bescheibenTurnstileExpired = () => {
    const input = document.querySelector('[data-turnstile-token]');
    if (input) input.value = '';
  };

  const form = document.querySelector('[data-diagnostic-form]');
  if (form) {
    const status = form.querySelector('[data-form-status]');
    const submit = form.querySelector('button[type="submit"]');
    const startedAt = form.querySelector('[data-started-at]');
    const phoneField = form.querySelector('[data-phone-field]');
    const phone = form.querySelector('[name="phone"]');
    startedAt.value = String(Date.now());

    const showStatus = (message, state = 'error') => {
      status.textContent = message;
      status.dataset.state = state;
      status.hidden = false;
      status.focus();
    };
    const clearErrors = () => {
      form.querySelectorAll('[aria-invalid="true"]').forEach((field) => field.removeAttribute('aria-invalid'));
      form.querySelectorAll('[data-error-for]').forEach((message) => { message.textContent = ''; });
    };
    const syncPhone = () => {
      const preference = form.querySelector('[name="contactPreference"]:checked')?.value;
      const visible = preference === 'whatsapp';
      phoneField.hidden = !visible;
      phone.required = visible;
      if (!visible) phone.value = '';
    };
    form.querySelectorAll('[name="contactPreference"]').forEach((control) => control.addEventListener('change', syncPhone));
    syncPhone();

    form.addEventListener('submit', async (event) => {
      event.preventDefault();
      clearErrors();
      if (!form.reportValidity()) return;
      if (form.dataset.turnstileConfigured !== 'true' && !form.querySelector('[data-turnstile-token]').value) {
        showStatus('A proteção do formulário ainda não está configurada neste ambiente.', 'fatal');
        return;
      }

      submit.disabled = true;
      submit.setAttribute('aria-busy', 'true');
      status.hidden = true;
      const payload = Object.fromEntries(new FormData(form).entries());
      // The widget adds its own field; the API accepts our callback token only.
      delete payload['cf-turnstile-response'];
      let delivered = false;
      try {
        const response = await fetch('/api/diagnostico', {
          method: 'POST',
          headers: { 'content-type': 'application/json' },
          body: JSON.stringify(payload),
        });
        const result = await response.json().catch(() => ({}));
        if (!response.ok) {
          if (result.errors) Object.entries(result.errors).forEach(([field, message]) => {
            const control = form.querySelector(`[name="${CSS.escape(field)}"]`);
            const output = form.querySelector(`[data-error-for="${CSS.escape(field)}"]`);
            control?.setAttribute('aria-invalid', 'true');
            if (output) output.textContent = String(message);
          });
          showStatus(result.message || 'Não foi possível enviar. Revise os campos e tente novamente.');
          return;
        }
        form.reset();
        delivered = true;
        syncPhone();
        startedAt.value = String(Date.now());
        showStatus('Contexto recebido. Vamos analisar as informações antes de continuar a conversa.', 'success');
        form.querySelectorAll('.field, fieldset, .cf-turnstile, .captcha-note, .form-submit').forEach((element) => { element.hidden = true; });
      } catch {
        showStatus('A conexão falhou. Seu texto foi preservado; tente novamente em alguns minutos.');
      } finally {
        if (!delivered) {
          // Verification consumes the token even when delivery subsequently fails.
          form.querySelector('[data-turnstile-token]').value = '';
          const widget = form.querySelector('.cf-turnstile');
          if (widget && window.turnstile) window.turnstile.reset(widget);
        }
        submit.disabled = false;
        submit.removeAttribute('aria-busy');
      }
    });
  }
</script>
